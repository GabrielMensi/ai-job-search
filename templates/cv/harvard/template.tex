%% Harvard-style CV template
%% Compile with: lualatex -interaction=nonstopmode <file>.tex
%% Profile-agnostic skeleton - fill [PLACEHOLDER] tokens with real data.
%% Classic centered-header, rule-under-heading resume layout. No color, no icons.
%% Font: Bitstream Charter. Hard page limit: 1 page.

\documentclass[10.5pt,a4paper]{article}

\usepackage[margin=0.75in]{geometry}
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{charter}
\usepackage{titlesec}
\usepackage{enumitem}
\usepackage{hyperref}
\hypersetup{
    colorlinks=true,
    linkcolor=black,
    urlcolor=black,
    pdftitle={[YOUR_NAME] - CV},
    pdfpagemode=UseNone,
}

\pagestyle{empty}
\setlength{\parindent}{0pt}

% Section headings: bold, full-width rule immediately beneath (\titlerule, never raw \hrule --
% \hrule is vertical-mode-only and renders as a strikethrough through the heading text if used
% inline here).
\titleformat{\section}{\normalsize\bfseries}{}{0pt}{}[{\titlerule[0.6pt]}]
\titlespacing*{\section}{0pt}{9pt}{5pt}

% Tight bullet lists - no inter-item \vspace (produces uneven gaps), rely on native spacing
\setlist[itemize]{leftmargin=14pt, itemsep=1pt, topsep=1pt, parsep=0pt}

% Entry header lines: bold org/institution (left) + location (right);
% italic role/degree (left) + dates (right), on the next line.
\newcommand{\cvheaderline}[2]{\textbf{#1} \hfill #2\\}
\newcommand{\cvsubheaderline}[2]{\textit{#1} \hfill \textit{#2}\\[1pt]}

% Underlined link text (colorlinks is black, so underline is what signals "clickable").
\newcommand{\cvlink}[2]{\href{#1}{\underline{#2}}}

\begin{document}

\end{center}
\vspace{2pt}
